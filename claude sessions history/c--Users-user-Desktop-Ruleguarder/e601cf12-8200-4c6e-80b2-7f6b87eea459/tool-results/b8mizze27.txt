    31	    findings: dict[int, list[Finding]] = {}

    34	    if llm_rules:

    36	            raise RuleValidationError("AI rules need an LLM provider, but none is configured.")
    37	        findings.update(

    39	                llm,

    41	                blocks,
    42	            )
    43	        )
    44	
    45	    for rule in rules:
    46	        rt = RULE_TYPES.get(rule.type)
    47	        if rt is None or rt.check is None:
    48	            continue

    50	
    51	    by_id = {b.id: b for b in blocks}
    52	    violations: list[Violation] = []

    55	            block = by_id.get(f.block_id) if f.block_id is not None else None
    56	            violations.append(

    59	                    rule_id=rule.id,
    60	                    rule_name=rule.name,

    62	                    message=f.message,
    63	                    block_id=f.block_id,

    65	                    end=f.end,

    67	                    location=_location(block),

    72	    summary = ReportSummary(

    80	def _excerpt(block: Block | None, start: int | None, end: int | None) -> str:
    81	    if block is None:
    82	        return ""
    83	    if start is None or end is None:
    84	        return block.text[: _EXCERPT_CONTEXT * 2]
    85	    lo = max(0, start - _EXCERPT_CONTEXT)
    86	    hi = min(len(block.text), end + _EXCERPT_CONTEXT)

     1	"""Rule types.
     2	
     3	The main rule type is "llm": a natural-language instruction that the LLM checks
     4	(e.g. "the document must not reveal numeric performance figures of the system").
     5	The other types are deterministic, free and exact - use them for formatting rules
     6	(fonts, sizes) that an LLM cannot see, or for plain forbidden words.
     7	
     8	To add a new deterministic type: define a params model + a checker function (here, or in

    96	def _check_forbidden_text(params: TextPatternParams, blocks: list[Block]) -> Iterable[Finding]:
    97	    pattern = params.compiled()
    98	    for block in blocks:
    99	        for m in pattern.finditer(block.text):
   100	            if m.end() == m.start():
   101	                continue
   102	            yield Finding(f'Forbidden text "{m.group(0)}"', block.id, m.start(), m.end())

   105	def _check_required_text(params: TextPatternParams, blocks: list[Block]) -> Iterable[Finding]:
   106	    pattern = params.compiled()
   107	    if not any(pattern.search(b.text) for b in blocks):
   108	        yield Finding(f'Required text "{params.pattern}" was not found anywhere in the document')

   111	def _check_max_sentence_words(params: MaxWordsParams, blocks: list[Block]) -> Iterable[Finding]:
   112	    for block in blocks:
   113	        for m in _SENTENCE_RE.finditer(block.text):
   114	            words = len(_WORD_RE.findall(m.group(0)))
   115	            if words > params.max_words:
   116	                # trim leading whitespace so the highlight starts on the first word
   117	                lead = len(m.group(0)) - len(m.group(0).lstrip())
   118	                yield Finding(
   119	                    f"Sentence has {words} words (max {params.max_words})",
   120	                    block.id,
   121	                    m.start() + lead,
   122	                    m.end(),
   123	                )

   126	def _check_max_paragraph_words(params: MaxWordsParams, blocks: list[Block]) -> Iterable[Finding]:
   127	    for block in blocks:
   128	        words = len(_WORD_RE.findall(block.text))
   129	        if words > params.max_words:
   130	            yield Finding(
   131	                f"Paragraph has {words} words (max {params.max_words})",
   132	                block.id,
   133	                0,
   134	                len(block.text),
   135	            )

   138	def _merge_run_findings(
   139	    blocks: list[Block], bad_value: Callable[[Run], Any], describe: Callable[[Any], str]
   140	) -> Iterable[Finding]:
   141	    """Emit one finding per stretch of consecutive runs sharing the same offending value."""
   142	    for block in blocks:
   143	        span: tuple[int, int, Any] | None = None
   144	        for run in block.runs:
   145	            value = bad_value(run) if run.text.strip() else None
   146	            if span and value is not None and value == span[2] and run.start == span[1]:
   147	                span = (span[0], run.end, value)
   148	                continue
   149	            if span:
   150	                yield Finding(describe(span[2]), block.id, span[0], span[1])
   151	            span = (run.start, run.end, value) if value is not None else None
   152	        if span:
   153	            yield Finding(describe(span[2]), block.id, span[0], span[1])

   156	def _check_allowed_fonts(params: AllowedFontsParams, blocks: list[Block]) -> Iterable[Finding]:
   157	    allowed = {f.strip().lower() for f in params.fonts}
   158	    return _merge_run_findings(
   159	        blocks,

     1	"""Run natural-language ("llm") rules over a parsed document.
     2	
     3	The model gets the document as numbered blocks and returns (rule, block, verbatim quote).
     4	We then locate the quote in the block text ourselves, so highlights always point at
     5	real text - a quote the model made up can't produce a bogus highlight.
     6	"""
     7	
     8	import re
     9	from concurrent.futures import ThreadPoolExecutor
    10	
    11	from ..config import get_settings
    12	from ..llm import LlmBlock, LlmProvider, LlmRule

    16	# Characters the model may silently swap (Hebrew gershayim/geresh vs ASCII quotes, dashes, NBSP...)
    17	_EQUIVALENT = str.maketrans(
    18	    {
    19	        "״": '"',
    20	        "“": '"',
    21	        "”": '"',
    22	        "„": '"',  # gershayim, curly quotes
    23	        "׳": "'",
    24	        "‘": "'",
    25	        "’": "'",
    26	        "`": "'",  # geresh, curly apostrophes
    27	        "–": "-",
    28	        "—": "-",
    29	        "־": "-",  # en/em dash, maqaf
    30	        " ": " ",  # no-break space
    31	        "‏": "",
    32	        "‎": "",
    33	        "​": "",  # RLM, LRM, zero-width space
    34	    }
    35	)
    36	_WS = re.compile(r"\s+")

    39	def check_llm_rules(provider: LlmProvider, rules: list[LlmRule], blocks: list[Block]) -> dict[int, list[Finding]]:
    40	    """Returns findings grouped by rule id."""
    41	    settings = get_settings()
    42	    llm_blocks = [LlmBlock(b.id, b.label, b.text) for b in blocks if b.text.strip()]
    43	    chunks = _chunk(llm_blocks, settings.llm_chunk_chars)
    44	    by_id = {b.id: b for b in blocks}
    45	    rule_ids = {r.id for r in rules}
    46	
    47	    with ThreadPoolExecutor(max_workers=max(1, settings.llm_max_parallel)) as pool:
    48	        results = list(pool.map(lambda chunk: provider.find_violations(rules, chunk), chunks))
    49	
    50	    findings: dict[int, list[Finding]] = {r.id: [] for r in rules}
    51	    seen: set[tuple[int, int | None, int | None, int | None]] = set()
    52	    doc_level_reported: set[int] = set()
    53	
    54	    for chunk, violations in zip(chunks, results, strict=True):
    55	        chunk_ids = [b.id for b in chunk]
    56	        for v in violations:
    57	            if v.rule_id not in rule_ids:
    58	                continue
    59	
    60	            if v.block_id is None:
    61	                # With several chunks, every chunk may claim "X is missing"; keep one.
    62	                if v.rule_id in doc_level_reported:
    63	                    continue
    64	                doc_level_reported.add(v.rule_id)
    65	                findings[v.rule_id].append(Finding(v.explanation))
    66	                continue
    67	
    68	            block_id, span = _locate(v.quote, v.block_id, chunk_ids, by_id)
    69	            if block_id is None:
    70	                block_id, span = v.block_id if v.block_id in by_id else None, None
    71	            start, end = span if span else (None, None)
    72	
    73	            key = (v.rule_id, block_id, start, end)
    74	            if key in seen:
    75	                continue
    76	            seen.add(key)
    77	
    78	            message = v.explanation
    79	            if span is None and v.quote:
    80	                message += f' (quoted: "{v.quote}")'
    81	            findings[v.rule_id].append(Finding(message, block_id, start, end))
    82	
    83	    return findings

    86	def _chunk(blocks: list[LlmBlock], max_chars: int) -> list[list[LlmBlock]]:
    87	    chunks: list[list[LlmBlock]] = []
    88	    current: list[LlmBlock] = []
    89	    size = 0
    90	    for b in blocks:
    91	        length = len(b.text) + len(b.label) + 10
    92	        if current and size + length > max_chars:
    93	            chunks.append(current)
    94	            current, size = [], 0
    95	        current.append(b)
    96	        size += length
    97	    if current or not chunks:
    98	        chunks.append(current)
    99	    return chunks
   100	
   101	
   102	def _locate(
   103	    quote: str, block_id: int, chunk_ids: list[int], by_id: dict[int, Block]
   104	) -> tuple[int | None, tuple[int, int] | None]:
   105	    """Find the quote in the claimed block first, then anywhere else in the same chunk."""
   106	    if not quote.strip():
   107	        return (block_id if block_id in by_id else None), None
   108	    candidates = [block_id] + [i for i in chunk_ids if i != block_id]
   109	    for cid in candidates:
   110	        block = by_id.get(cid)
   111	        if block is None:
   112	            continue
   113	        span = find_span(block.text, quote)
   114	        if span:
   115	            return cid, span
   116	    return None, None
   117	
   118	
   119	def find_span(text: str, quote: str) -> tuple[int, int] | None:
   120	    idx = text.find(quote)
   121	    if idx >= 0:
   122	        return idx, idx + len(quote)
   123	
   124	    norm_text, index_map = _normalize_with_map(text)
   125	    norm_quote, _ = _normalize_with_map(quote)
   126	    norm_quote = norm_quote.strip()
   127	    if not norm_quote:
   128	        return None
   129	    idx = norm_text.find(norm_quote)
   130	    if idx < 0:
   131	        idx = norm_text.lower().find(norm_quote.lower())
   132	    if idx < 0:
   133	        return None
   134	    return index_map[idx], index_map[idx + len(norm_quote) - 1] + 1
   135	
   136	
   137	def _normalize_with_map(s: str) -> tuple[str, list[int]]:
   138	    """Normalise look-alike characters and collapse whitespace, remembering original indices."""
   139	    out: list[str] = []
   140	    index_map: list[int] = []
   141	    for i, ch in enumerate(s):
   142	        ch = ch.translate(_EQUIVALENT)
   143	        if not ch:
   144	            continue
   145	        if _WS.fullmatch(ch):
   146	            if out and out[-1] == " ":
   147	                continue
   148	            ch = " "
   149	        out.append(ch)
   150	        index_map.append(i)
   151	    return "".join(out), index_map

    84	        response = self._generate(_render_prompt(rules, blocks))
    85	
    86	        feedback = response.prompt_feedback
    87	        if feedback is not None and feedback.block_reason:
    88	            raise LlmError(f"Gemini blocked the request ({feedback.block_reason}).")
    89	
    90	        candidate = response.candidates[0] if response.candidates else None
    91	        finish = candidate.finish_reason if candidate else None
    92	        if finish == types.FinishReason.MAX_TOKENS:
    93	            raise LlmError("Gemini's answer was cut off; lower LLM_CHUNK_CHARS or raise LLM_MAX_TOKENS.")
    94	        if finish is not None and finish != types.FinishReason.STOP:
    95	            raise LlmError(f"Gemini stopped early ({finish}).")

    99	            try:

   101	            except ValidationError as exc:

   103	                raise LlmError("Gemini returned an invalid response.") from exc
   104

   106	            LlmViolation(

   108	                block_id=v.block_id if v.block_id >= 0 else None,

   110	                explanation=v.explanation,

   115	    def _generate(self, prompt: str) -> types.GenerateContentResponse:
   116	        """Try each model in order. Overload/timeout => next model; rate limit => back off, then next."""
   117	        failures: list[str] = []
   118	        for model in self._models:
   119	            for attempt in range(self._max_retries + 1):
   120	                started = time.monotonic()
   121	                try:
   122	                    response = self._client.models.generate_content(model=model, contents=prompt, config=self._config)
   123	                    log.info("Gemini %s answered in %.1fs", model, time.monotonic() - started)
   124	                    return response
   125	                except errors.APIError as exc:
   126	                    if exc.code == _RATE_LIMITED and attempt < self._max_retries:
   127	                        delay = min(2**attempt * 2 + random.uniform(0, 1), 60)
   128	                        log.info("Gemini %s rate limited, retrying in %.1fs", model, delay)
   129	                        time.sleep(delay)
   130	                        continue
   131	                    if exc.code == _RATE_LIMITED or exc.code in _OVERLOADED:
   132	                        log.warning("Gemini %s unavailable (%s), trying next model", model, exc.code)
   133	                        failures.append(f"{model}: {exc.code} {exc.message}")
   134	                        break
   135	                    raise LlmError(f"Gemini request failed ({exc.code}): {exc.message}") from exc
   136	                except httpx.TimeoutException:
   137	                    log.warning("Gemini %s timed out after %.0fs, trying next model", model, time.monotonic() - started)
   138	                    failures.append(f"{model}: timed out")
   139	                    break
   140	                except httpx.HTTPError as exc:
   141	                    raise LlmError(f"Could not reach the Gemini API: {exc}") from exc
   142	        raise LlmError("All Gemini models are unavailable right now - " + "; ".join(failures))

    27	def parse_docx(source: str | IO[bytes]) -> list[Block]:
    28	    try:
    29	        doc = open_docx(source)
    30	    except Exception as exc:  # python-docx raises a variety of zip/xml errors
    31	        raise DocxParseError(f"Could not read DOCX file: {exc}") from exc
    32	
    33	    defaults = _doc_defaults(doc)
    34	    blocks: list[Block] = []
    35	    current_heading: str | None = None
    36	    paragraph_no = 0
    37	    table_no = 0
    38	
    39	    for item in _iter_block_items(doc):
    40	        if isinstance(item, Paragraph):
    41	            paragraph_no += 1
    42	            block = _paragraph_to_block(
    43	                item,
    44	                block_id=len(blocks),
    45	                defaults=defaults,
    46	                heading_context=current_heading,
    47	                label=f"Paragraph {paragraph_no}",

    50	            if block.kind == "heading" and block.text.strip():
    51	                current_heading = block.text.strip()
    52	                block.heading_context = None
    53	            blocks.append(block)
    54	            continue
    55	
    56	        table_no += 1
    57	        # Holds the <w:tc> elements themselves: that keeps lxml's proxies alive, so the
    58	        # same cell always maps to the same object (id() of a freed proxy gets reused).
    59	        seen_cells: set[Any] = set()
    60	        for r, row in enumerate(item.rows, start=1):
    61	            for c, cell in enumerate(row.cells, start=1):
    62	                # Merged cells are returned once per grid position; only emit them once.
    63	                if cell._tc in seen_cells:
    64	                    continue
    65	                seen_cells.add(cell._tc)
    66	                for p in cell.paragraphs:
    67	                    block = _paragraph_to_block(
    68	                        p,
    69	                        block_id=len(blocks),
    70	                        defaults=defaults,
    71	                        heading_context=current_heading,
    72	                        label=f"Table {table_no}, row {r}, column {c}",
    73	                    )
    74	                    block.kind = "table_cell"
    75	                    block.table_index, block.row, block.col = table_no, r, c
    76	                    blocks.append(block)

   147	def _style_attr(style: Any, attr: str) -> Any:
   148	    """Walk the style inheritance chain until a value for font.<attr> is found."""
   149	    while style is not None:
   150	        font = getattr(style, "font", None)
   151	        value = getattr(font, attr, None) if font is not None else None
   152	        if value is not None:
   153	            return value
   154	        style = getattr(style, "base_style", None)
   155	    return None

   159	    if direct is not None:
   160	        return direct
   161	    value = _style_attr(run_style, attr)
   162	    if value is None:
   163	        value = _style_attr(para_style, attr)
   164	    return None if value is None else bool(value)

   171	def _doc_defaults(doc: DocxDocument) -> dict[str, Any]:
   172	    """Read <w:docDefaults> so runs without explicit/style formatting still get a font + size."""

   182	            size_pt = int(sz.get(qn("w:val"))) / 2  # stored in half-points
