    15	export class ApiError extends Error {

    24	async function send(path: string, init?: RequestInit): Promise<Response> {
    25	  const res = await fetch(BASE + path, init);
    26	  if (!res.ok) {
    27	    let message = res.statusText;
    28	    try {
    29	      const body = (await res.json()) as { detail?: unknown };
    30	      message = typeof body.detail === "string" ? body.detail : JSON.stringify(body.detail);
    31	    } catch {
    32	      /* non-JSON error body */
    33	    }
    34	    throw new ApiError(res.status, message);
    35	  }
    36	  return res;
    37	}
    38	
    39	async function request<T>(path: string, init?: RequestInit): Promise<T> {
    40	  const res = await send(path, init);
    41	  return (await res.json()) as T;
    42	}
    43	
    44	async function requestNoContent(path: string, init?: RequestInit): Promise<void> {
    45	  await send(path, init);
    46	}

   233	export function groupTemplates(templates: readonly RuleTemplate[]): TemplateGroup[] {
   234	  const groups: TemplateGroup[] = CATEGORY_ORDER.map((category) => ({
   235	    category,
   236	    label: CATEGORY_LABEL[category],

   240	  return groups.filter((g) => g.templates.length > 0);

   362	/** Loads the latest report for a document (or the bare document if never checked) and runs checks. */
   363	export function useDocumentReport(documentId: number | null) {
   364	  const [state, setState] = useState<DocumentReportState>(EMPTY);
   365	  const [loadedId, setLoadedId] = useState<number | null>(null);
   366	
   367	  // Reset during render when the selection changes (avoids setState inside an effect)
   368	  if (loadedId !== documentId) {
   369	    setLoadedId(documentId);
   370	    setState({ ...EMPTY, loading: documentId !== null });
   371	  }
   372	
   373	  useEffect(() => {
   374	    if (documentId === null) return;
   375	    let cancelled = false;
   376	    load(documentId)
   377	      .then((next) => !cancelled && setState({ ...EMPTY, ...next }))
   378	      .catch((err: unknown) => !cancelled && setState({ ...EMPTY, error: errorMessage(err) }));
   379	    return () => {
   380	      cancelled = true;
   381	    };
   382	  }, [documentId]);
   383	
   384	  const runCheck = useCallback(async () => {
   385	    if (documentId === null) return;
   386	    setState((s) => ({ ...s, checking: true, error: null }));
   387	    try {
   388	      const report = await api.check(documentId);
   389	      setState({
   390	        ...EMPTY,
   391	        document: report.document,
   392	        violations: report.violations,
   393	        checkedAt: report.checked_at,
   394	      });
   395	    } catch (err) {
   396	      setState((s) => ({ ...s, checking: false, error: errorMessage(err) }));
   397	    }
   398	  }, [documentId]);
   399	
   400	  return { ...state, runCheck };
   401	}

   408	    if (!(err instanceof ApiError && err.status === 404)) throw err;
   409	    return { document: await api.document(documentId) };
   410	  }

   496	/** Uploaded documents: list, upload and delete. Failures are reported through `error`. */
   497	export function useDocuments() {
   498	  const [state, setState] = useState<DocumentsState>({ documents: [], loading: true, uploading: false, error: null });
   499	
   500	  useEffect(() => {
   501	    let cancelled = false;
   502	    api
   503	      .documents()
   504	      .then((documents) => !cancelled && setState((s) => ({ ...s, documents, loading: false })))
   505	      .catch((err: unknown) => !cancelled && setState((s) => ({ ...s, loading: false, error: errorMessage(err) })));
   506	    return () => {
   507	      cancelled = true;
   508	    };
   509	  }, []);
   510	
   511	  const reload = useCallback(async () => {
   512	    try {
   513	      const documents = await api.documents();
   514	      setState((s) => ({ ...s, documents, error: null }));
   515	    } catch (err) {
   516	      setState((s) => ({ ...s, error: errorMessage(err) }));
   517	    }
   518	  }, []);
   519	
   520	  /** Resolves to the uploaded document, or null when the upload failed. */
   521	  const upload = useCallback(
   522	    async (file: File): Promise<DocumentFull | null> => {
   523	      setState((s) => ({ ...s, uploading: true, error: null }));
   524	      try {
   525	        const doc = await api.uploadDocument(file);
   526	        setState((s) => ({ ...s, uploading: false }));
   527	        await reload();
   528	        return doc;
   529	      } catch (err) {
   530	        setState((s) => ({ ...s, uploading: false, error: errorMessage(err) }));
   531	        return null;
   532	      }
   533	    },
   534	    [reload],
   535	  );
   536	
   537	  /** Resolves to true when the document was deleted. */
   538	  const remove = useCallback(
   539	    async (id: number): Promise<boolean> => {
   540	      try {
   541	        await api.deleteDocument(id);
   542	        await reload();
   543	        return true;
   544	      } catch (err) {
   545	        setState((s) => ({ ...s, error: errorMessage(err) }));
   546	        return false;

   655	/**
   656	 * Rule templates (`GET /api/rules/templates`) and the sample set (`POST /api/rules/samples`).
   657	 * `onRulesChanged` is called after samples were loaded so the rules list refreshes.
   658	 */
   659	export function useRuleTemplates(onRulesChanged: () => unknown) {
   660	  const [state, setState] = useState<TemplatesState>({ templates: [], status: "loading", error: null });
   661	  const [samples, setSamples] = useState<SamplesState>({ loading: false, message: null, error: null });
   662	  const [attempt, setAttempt] = useState(0);
   663	
   664	  useEffect(() => {
   665	    let cancelled = false;
   666	    api
   667	      .ruleTemplates()
   668	      .then((templates) => !cancelled && setState({ templates, status: "ready", error: null }))
   669	      .catch((err: unknown) => {
   670	        if (cancelled) return;
   671	        setState(
   672	          isMissingEndpoint(err)
   673	            ? { templates: [], status: "unavailable", error: null }
   674	            : { templates: [], status: "error", error: errorMessage(err) },
   675	        );
   676	      });
   677	    return () => {
   678	      cancelled = true;
   679	    };
   680	  }, [attempt]);
   681	
   682	  const retry = useCallback(() => {
   683	    setState((s) => ({ ...s, status: "loading", error: null }));
   684	    setAttempt((n) => n + 1);
   685	  }, []);
   686	
   687	  const loadSamples = useCallback(async () => {
   688	    setSamples({ loading: true, message: null, error: null });
   689	    try {
   690	      const result = await api.loadSampleRules();
   691	      setSamples({ loading: false, message: sampleResultMessage(result), error: null });
   692	      await onRulesChanged();
   693	    } catch (err) {
   694	      setSamples({
   695	        loading: false,
   696	        message: null,
   697	        error: isMissingEndpoint(err) ? SAMPLES_UNAVAILABLE : errorMessage(err),
   698	      });
   699	    }
   700	  }, [onRulesChanged]);
   701	
   702	  const dismissSamples = useCallback(() => setSamples({ loading: false, message: null, error: null }), []);
   703	
   704	  return { ...state, retry, samples, loadSamples, dismissSamples };
   705	}

   790	export function useRules() {

   792	
   793	  useEffect(() => {
   794	    let cancelled = false;
   795	    Promise.all([api.rules(), api.ruleTypes()])
   796	      .then(([rules, types]) => !cancelled && setState({ rules, types, loading: false, error: null }))
   797	      .catch((err: unknown) => !cancelled && setState((s) => ({ ...s, loading: false, error: errorMessage(err) })));
   798	    return () => {
   799	      cancelled = true;
   800	    };
   801	  }, []);
   802	
   803	  const reload = useCallback(async () => {
   804	    try {
   805	      const rules = await api.rules();
   806	      setState((s) => ({ ...s, rules, error: null }));
   807	    } catch (err) {
   808	      setState((s) => ({ ...s, error: errorMessage(err) }));
   809	    }
   810	  }, []);
   811	
   812	  const createRule = useCallback(
   813	    async (input: RuleCreate) => {
   814	      await api.createRule(input);
   815	      await reload();
   816	    },
   817	    [reload],
   818	  );
   819	
   820	  const updateRule = useCallback(
   821	    async (id: number, changes: RuleUpdate) => {
   822	      await api.updateRule(id, changes);
   823	      await reload();
   824	    },
   825	    [reload],
   826	  );
   827	
   828	  const guarded = useCallback(
   829	    async (action: () => Promise<unknown>) => {
   830	      try {
   831	        await action();
   832	        await reload();
   833	      } catch (err) {
   834	        setState((s) => ({ ...s, error: errorMessage(err) }));

   841	    (rule: Rule) => guarded(() => api.updateRule(rule.id, { enabled: !rule.enabled })),
   842	    [guarded],
   843	  );
   844	  const deleteRule = useCallback((rule: Rule) => guarded(() => api.deleteRule(rule.id)), [guarded]);

   860	/**
   861	 * Severity chips + rule filter over a report's violations.
   862	 * Chip counts reflect the rule filter, so they always match what a chip would show.
   863	 * Reset by giving the owning component a new `key` per report.
   864	 */
   865	export function useViolationFilters(violations: readonly Violation[]) {
   866	  const [severities, setSeverities] = useState<ReadonlySet<Severity>>(ALL_SEVERITIES);
   867	  const [ruleId, setRuleId] = useState<number | null>(null);
   868	
   869	  const rules = useMemo(() => ruleOptions(violations), [violations]);
   870	  // A stale rule filter (rule no longer in this report) behaves like "all rules"
   871	  const effectiveRuleId = ruleId !== null && rules.some((r) => r.ruleId === ruleId) ? ruleId : null;
   872	
   873	  const counts = useMemo(
   874	    () => countBySeverity(filterViolations(violations, { severities: ALL_SEVERITIES, ruleId: effectiveRuleId })),
   875	    [violations, effectiveRuleId],
   876	  );
   877	  const visible = useMemo(
   878	    () => documentLevelFirst(filterViolations(violations, { severities, ruleId: effectiveRuleId })),
   879	    [violations, severities, effectiveRuleId],
   880	  );
   881	
   882	  const toggleSeverity = useCallback((s: Severity) => setSeverities((current) => toggle(current, s)), []);
   883	  const clear = useCallback(() => {
   884	    setSeverities(ALL_SEVERITIES);
   885	    setRuleId(null);
   886	  }, []);
   887	
   888	  const isFiltered = effectiveRuleId !== null || severities.size !== ALL_SEVERITIES.size;
   889	
   890	  return { severities, toggleSeverity, ruleId: effectiveRuleId, setRuleId, rules, counts, visible, isFiltered, clear };

   896	export interface Flash {
   897	  blockId: number;
   898	  /** Changes on every click, so a keyed overlay remounts and the CSS animation restarts. */
   899	  nonce: number;

   902	/**
   903	 * The selected ("active") violation and the paragraph flash.
   904	 * - From the list: scroll the paragraph into view (smooth), mark it active and flash it for FLASH_MS.
   905	 * - From the document (clicking a highlight): mark it active and scroll its card into view in the list.
   906	 * Document-level violations (block_id null) only become active.
   907	 */
   908	export function useViolationFocus() {
   909	  const [activeId, setActiveId] = useState<string | null>(null);
   910	  const [flash, setFlash] = useState<Flash | null>(null);
   911	  const blocks = useRef(new Map<number, HTMLElement>());
   912	  const items = useRef(new Map<string, HTMLElement>());
   913	  const timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined);
   914	  const nonce = useRef(0);
   915	
   916	  useEffect(() => () => clearTimeout(timer.current), []);
   917	
   918	  const selectFromList = useCallback((v: Violation) => {
   919	    setActiveId(v.id);
   920	    if (v.block_id === null) return;
   921	    const blockId = v.block_id;
   922	    blocks.current.get(blockId)?.scrollIntoView({ behavior: scrollBehavior(), block: "center" });
   923	    nonce.current += 1;
   924	    setFlash({ blockId, nonce: nonce.current });
   925	    clearTimeout(timer.current);
   926	    timer.current = setTimeout(() => setFlash(null), FLASH_MS);
   927	  }, []);
   928	
   929	  const selectFromDocument = useCallback((v: Violation) => {
   930	    setActiveId(v.id);
   931	    items.current.get(v.id)?.scrollIntoView({ behavior: scrollBehavior(), block: "nearest" });
   932	  }, []);
   933	
   934	  const registerBlock = useCallback(
   935	    (id: number) => (el: HTMLElement | null) => {
   936	      if (el) blocks.current.set(id, el);
   937	      else blocks.current.delete(id);
   938	    },
   939	    [],
   940	  );
   941	
   942	  const registerItem = useCallback(
   943	    (id: string) => (el: HTMLElement | null) => {
   944	      if (el) items.current.set(id, el);
   945	      else items.current.delete(id);
   946	    },
   947	    [],
   948	  );
   949	
   950	  return { activeId, flash, selectFromList, selectFromDocument, registerBlock, registerItem };
   951	}
